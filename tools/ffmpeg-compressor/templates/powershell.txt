# ff-tool: Windows PowerShell 5.1+ / PowerShell 7. Save as UTF-8 with BOM.
# Run: & .\ff-tool.ps1  (requires ffmpeg/libx264 and ffprobe in PATH)
& {
$ErrorActionPreference = 'Stop'
$PSNativeCommandUseErrorActionPreference = $false
foreach ($name in @('ffmpeg','ffprobe')) { if (-not (Get-Command $name -ErrorAction SilentlyContinue)) { throw "Missing required command: $name" } }
$targetBytes = [long]@@TARGET_BYTES@@
$ratio = [double]@@RATIO@@
$audioMode = @@AUDIO_MODE@@
$audioKbps = @@AUDIO_KBPS@@
$preset = @@PRESET@@
$filter = @@FILTER@@
$fps = @@FPS@@
$jobs = @(
@@JOBS@@
)
$succeeded = 0; $failed = 0
foreach ($job in $jobs) {
  $workDir = $null
  $ownedTemp = $false
  try {
    $inputFile = [IO.Path]::GetFullPath($job.InputFile)
    $outputFile = [IO.Path]::GetFullPath($job.OutputFile)
    Write-Host "`n[START] $inputFile"
    if (-not [IO.File]::Exists($inputFile)) { throw 'Source does not exist.' }
    if (Test-Path -LiteralPath $outputFile) { throw "Output exists; choose a new name: $outputFile" }
    $outDir = [IO.Path]::GetDirectoryName($outputFile)
    if (-not [IO.Directory]::Exists($outDir)) { throw 'Output directory does not exist.' }
    $probeArgs = @('-v','error','-show_entries','format=duration:stream=codec_type,width,height','-of','json','-i',$inputFile)
    $raw = & ffprobe @probeArgs
    if ($LASTEXITCODE -ne 0) { throw 'ffprobe failed.' }
    $probe = ($raw -join "`n") | ConvertFrom-Json
    $videos = @($probe.streams | Where-Object { $_.codec_type -eq 'video' })
    if ($videos.Count -eq 0 -or $videos[0].width -lt 2 -or $videos[0].height -lt 2) { throw 'No usable video stream.' }
    $seconds = [double]0
    $valid = [double]::TryParse([string]$probe.format.duration, [Globalization.NumberStyles]::Float, [Globalization.CultureInfo]::InvariantCulture, [ref]$seconds)
    if (-not $valid -or $seconds -le 0 -or [double]::IsNaN($seconds) -or [double]::IsInfinity($seconds)) { $seconds = [double]$job.ManualSeconds }
    if ($seconds -le 0) { throw 'Duration unavailable; supply a valid manual duration.' }
    $actualAudio = 0
    if ($audioMode -eq 'keep' -and @($probe.streams | Where-Object { $_.codec_type -eq 'audio' }).Count -gt 0) { $actualAudio = $audioKbps }
    $videoKbps = [math]::Floor(($targetBytes * $ratio * 8 / $seconds - $actualAudio * 1000) / 1000)
    if ($videoKbps -lt 16) { throw 'Insufficient video budget (<16 kbps). Increase target or remove audio.' }
    $videoKbpsText = $videoKbps.ToString('0', [Globalization.CultureInfo]::InvariantCulture)
    Write-Host "Duration $seconds s | video $videoKbpsText kbps | audio $actualAudio kbps"
    $workDir = Join-Path $outDir ('.ff-tool-' + [guid]::NewGuid().ToString('N'))
    if ([IO.Directory]::Exists($workDir)) { throw 'Temporary directory collision; rerun the script.' }
    $null = [IO.Directory]::CreateDirectory($workDir)
    $ownedTemp = $true
    $commonArgs = @('-nostdin','-hide_banner','-loglevel','error','-i',$inputFile,'-map','0:v:0','-map_metadata','-1','-map_chapters','-1','-c:v','libx264','-b:v',($videoKbpsText+'k'),'-preset',$preset,'-pix_fmt','yuv420p','-vf',$filter,'-passlogfile',(Join-Path $workDir 'pass'))
    if ($fps -ne 'source') { $commonArgs += @('-r',$fps) }
    $firstArgs = $commonArgs + @('-pass','1','-an','-f','null','NUL')
    & ffmpeg @firstArgs
    if ($LASTEXITCODE -ne 0) { throw 'Pass 1 failed; pass 2 was not started.' }
    $soundArgs = @('-an')
    if ($actualAudio -gt 0) { $soundArgs = @('-map','0:a:0','-c:a','aac','-b:a',([string]$actualAudio+'k'),'-ac','2') }
    $resultFile = Join-Path $workDir 'result.mp4'
    $secondArgs = $commonArgs + @('-pass','2') + $soundArgs + @('-movflags','+faststart','-n',$resultFile)
    & ffmpeg @secondArgs
    if ($LASTEXITCODE -ne 0) { throw 'Pass 2 failed.' }
    # .NET Move without an overwrite parameter fails if the destination exists.
    [IO.File]::Move($resultFile, $outputFile)
    $bytes = (Get-Item -LiteralPath $outputFile).Length
    Write-Host "[OK] $outputFile | $bytes bytes | target $targetBytes bytes"
    if ($bytes -gt $targetBytes) { throw 'OVERSIZE: output kept. Increase margin or reduce target; choose a NEW output name.' }
    $succeeded++
  } catch {
    $failed++
    Write-Warning ("[FAIL] " + $job.InputFile + ': ' + $_.Exception.Message)
  } finally {
    if ($ownedTemp -and $workDir -and [IO.Directory]::Exists($workDir)) { Remove-Item -LiteralPath $workDir -Recurse -Force }
  }
}
Write-Host "`nSummary: $succeeded succeeded / $failed failed or oversized."
if ($failed -gt 0) { throw "$failed job(s) failed or exceeded the size target." }
}
